  2# This is an EXUDYN example
  3#
  4# Details:  The HT parameters of the items (#2793): ObjectGround takes its frame as referencePosition and
  5#           referenceRotation or at once as referenceHT - a 4x4 matrix, its 16 values or an exu.HT. A parameter left
  6#           None is not given; GetObject lists all three, and such a dictionary can be added again; an HT and a part
  7#           given together must agree, else the item raises. SetObjectParameter of a part keeps the other part.
  8#           CreateGround and CreateRigidBody take referenceHT and initialHT for the position and rotation matrix;
  9#           for the three rigid body nodes they give the same coordinates, and an HT with one of its parts raises.
 10#           The rigid markers take localHT, the marker frame being the body or node frame times localHT: the rotation
 11#           matrix and the local angular velocity of MarkerBodyRigid, MarkerNodeRigid and MarkerKinematicTreeRigid; a node marker refuses a
 12#           translation; a joint whose markers carry its rotation in localHT moves as one with rotationMarker0/1.
 13#           ObjectKinematicTree takes its joint transformations and offsets as one list jointHTs and moves the same;
 14#           numpy reads an exu.HT as its 4x4 matrix, and the HT functions of rigidBodyUtilities and the robotics
 15#           classes take it.
 16#
 17# Author:   Johannes Gerstmayr
 18# Date:     2026-10-03
 19#
 20# Copyright:This file is part of Exudyn. Exudyn is free software. You can redistribute it and/or modify it under the terms of the Exudyn license. See 'LICENSE.txt' for more details.
 21#

150#a joint: the rotation in localHT of its markers or in rotationMarker0/1 of the joint gives the same motion
151def SwingingBody(useLocalHT, nodeMarker):
152    SCj = exu.SystemContainer()
153    mbsj = SCj.AddSystem()
154    gGround = mbsj.AddObject(ObjectGround())
155    oPendulum = mbsj.CreateRigidBody(inertia=inertia, referencePosition=[0.5, 0, 0], gravity=[0, -9.81, 0],
156                                     initialAngularVelocity=[0.2, 0.1, 0.])
157    Aaxis = RotXYZ2RotationMatrix([0.3, -0.4, 0.2]) #the joint axis z turned
158    HT0 = HomogeneousTransformation(Aaxis, [0, 0, 0])
159    HT1 = HomogeneousTransformation(Aaxis, [-0.5, 0, 0])
160    mGround = mbsj.AddMarker(MarkerBodyRigid(bodyNumber=gGround, localHT=HT0 if useLocalHT else None))
161    if nodeMarker: #the joint at the node of the body, its rotation in the node marker
162        mPendulum = mbsj.AddMarker(MarkerNodeRigid(nodeNumber=mbsj.GetObject(oPendulum)['nodeNumber'],
163                                                   localHT=HT0 if useLocalHT else None))
164        mGround = mbsj.AddMarker(MarkerBodyRigid(bodyNumber=gGround, localPosition=[0.5, 0, 0],
165                                                 localHT=None))
166        if useLocalHT:
167            mbsj.SetMarkerParameter(mGround, 'localHT', HomogeneousTransformation(Aaxis, [0.5, 0, 0]))
168    else:
169        mPendulum = mbsj.AddMarker(MarkerBodyRigid(bodyNumber=oPendulum, localHT=HT1 if useLocalHT else None,
170                                                   localPosition=None if useLocalHT else [-0.5, 0, 0]))
171    if useLocalHT:
172        mbsj.AddObject(ObjectJointGeneric(markerNumbers=[mGround, mPendulum], constrainedAxes=[1,1,1,1,1,0]))
173    else: #the deprecated rotationMarker0/1, compared with localHT; their warning is expected (#2873)
174        with warnings.catch_warnings():
175            warnings.simplefilter('ignore', DeprecationWarning)
176            mbsj.AddObject(ObjectJointGeneric(markerNumbers=[mGround, mPendulum], constrainedAxes=[1,1,1,1,1,0],
177                                              rotationMarker0=Aaxis, rotationMarker1=Aaxis))
178    mbsj.Assemble()
179    simulationSettings = exu.SimulationSettings()
180    simulationSettings.timeIntegration.newton.useModifiedNewton = False #Just for the test; modified Newton is usually faster
181    simulationSettings.timeIntegration.numberOfSteps = 200
182    simulationSettings.timeIntegration.endTime = 0.5
183    simulationSettings.timeIntegration.verboseMode = 0
184    mbsj.SolveDynamic(simulationSettings)
185    return mbsj.GetObjectOutputBody(oPendulum, OV.Position, [0.5, 0.1, 0.])

194#ObjectKinematicTree: jointHTs, the joint transformations and offsets as one list of HTs (#2798)
195def SwingingTree(useJointHTs):
196    SCk = exu.SystemContainer()
197    mbsk = SCk.AddSystem()
198    rotations = [RotXYZ2RotationMatrix([0.1, 0.2, 0.3]), RotXYZ2RotationMatrix([-0.3, 0.1, 0.])]
199    offsets = [[0., 0., 0.], [0.5, 0., 0.1]]
200    nTree = mbsk.AddNode(NodeGenericODE2(referenceCoordinates=[0., 0.], initialCoordinates=[0.3, -0.2],
201                                         initialCoordinates_t=[0., 0.], numberOfODE2Coordinates=2))
202    jointData = ({'jointHTs': [exu.HT(rotations[0], offsets[0]), HomogeneousTransformation(rotations[1], offsets[1])]}
203                 if useJointHTs else {'jointTransformations': exu.Matrix3DList(rotations), 'jointOffsets': exu.Vector3DList(offsets)})
204    oTree = mbsk.AddObject(ObjectKinematicTree(nodeNumber=nTree, jointTypes=[exu.JointType.RevoluteZ]*2, linkParents=[-1, 0],
205                                               linkInertiasCOM=exu.Matrix3DList([np.eye(3)*0.01]*2),
206                                               linkCOMs=exu.Vector3DList([[0.25, 0, 0]]*2), linkMasses=[1., 1.],
207                                               gravity=[0, -9.81, 0], **jointData))
208    mbsk.Assemble()
209    simulationSettings = exu.SimulationSettings()
210    simulationSettings.timeIntegration.newton.useModifiedNewton = False #Just for the test; modified Newton is usually faster
211    simulationSettings.timeIntegration.numberOfSteps = 100
212    simulationSettings.timeIntegration.endTime = 0.2
213    simulationSettings.timeIntegration.verboseMode = 0
214    mbsk.SolveDynamic(simulationSettings)
215    return (mbsk, oTree, mbsk.GetNodeOutput(nTree, exu.OutputVariableType.Coordinates))
