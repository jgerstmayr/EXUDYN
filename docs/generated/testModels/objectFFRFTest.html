155    n = int(len(v)/3) #number of nodes
156    sm = np.zeros((3*n,3))
157
158    for i in range(n):
159        off = 3*i
160        x=v[off+0]
161        y=v[off+1]
162        z=v[off+2]
163        mLoc = np.array([[0,-z,y],[z,0,-x],[-y,x,0]])
164        sm[off:off+3,:] = mLoc[:,:]
165
166    return sm

278    force = np.zeros(nODE2FFRF)

292    omega = np.array(list(omega3D)*nNodes)
293    omegaTilde = np.kron(np.eye(nNodes),omega3Dtilde)
294
295    #squared angul. vel. matrix:
296    omega3Dtilde2 = Skew(omega3D) @ Skew(omega3D)
297    omegaTilde2 = np.kron(np.eye(nNodes),omega3Dtilde2)
298
299    if True: #for rotordynamics, we assume rigid body motion with constant ang. vel.
300        #these 2 terms are computationally costly:
301        rfTilde = ComputeSkewMatrix(rF) #rfTilde
302        cF_tTilde = ComputeSkewMatrix(cF_t) 
303
304        fTrans = A @ (omega3Dtilde @ PhitTM @ rfTilde @ omega3D + 2*PhitTM @ cF_tTilde @ omega3D)
305        force[0:dim3D] = fTrans
306
307        fRot = -G.T@(omega3Dtilde @ rfTilde.T @ massMatrix @ rfTilde @ omega3D + 
308                        2*rfTilde.T @ massMatrix @ cF_tTilde @ omega3D)
309        force[dim3D:nODE2rigid] = fRot
310    
311    fFlex = -massMatrix @ (omegaTilde2 @ rF + 2*(omegaTilde @ cF_t))
312    force[nODE2rigid:] = fFlex
313
314    #add gravity:
315    if False:
316        fGrav = np.array(fGravRigid + list(PhitTM.T @ (A.T @ g)) ) #only local vector, without rotation
317        force += fGrav
318
319
320    return force
