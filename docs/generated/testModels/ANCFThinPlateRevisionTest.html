 46def StaticSettings(numberOfLoadSteps=numberOfTipLoadSteps):
 47    """Return simulation settings for the static tests (no solution file, sparse solver).
 48    @param numberOfLoadSteps: number of static load steps
 49    @return: exu.SimulationSettings"""
 50    simulationSettings = exu.SimulationSettings()
 51    simulationSettings.staticSolver.numberOfLoadSteps = numberOfLoadSteps
 52    simulationSettings.staticSolver.newton.relativeTolerance = 1e-10
 53    simulationSettings.staticSolver.newton.absoluteTolerance = 1e-3   #round-off floor of the residual (N)
 54    simulationSettings.staticSolver.verboseMode = 0
 55    simulationSettings.solution.file.write = False
 56    simulationSettings.linearSolver.solverType = exu.LinearSolverType.EigenSparse
 57    return simulationSettings

 60def CreateStrip(numberOfElements=4, integrationMode=0, alongY=False, clamp=True, thicknessField=None,
 61                massProportionalLoad=None, stiffnessProportionalDamping=0.):
 62    """Create a plate strip (stripLength x stripWidth, one element across the width) with ShellMesh, optionally
 63    clamped at the edge at the origin.
 64    @param numberOfElements: number of elements along the strip
 65    @param integrationMode: useReducedOrderIntegration of the elements
 66    @param alongY: True: strip axis along global y, False: along global x
 67    @param clamp: True: all 9 coordinates of the nodes at the origin edge are fixed
 68    @param thicknessField: function f(x, y) of the thickness, evaluated at the nodes (4 values per element)
 69    @param massProportionalLoad: a load per mass, e.g. gravity, on every element
 70    @param stiffnessProportionalDamping: Kelvin-Voigt damping coefficient of ShellMesh
 71    @return: (system container, mbs, dictionary of nodes, elements, edgeNodeNumbers and nodeId, list of tip node numbers)"""
 72    systemContainer = exu.SystemContainer()
 73    mbs = systemContainer.AddSystem()
 74    (lengthX, lengthY, nx, ny) = ((stripWidth, stripLength, 1, numberOfElements) if alongY
 75                                  else (stripLength, stripWidth, numberOfElements, 1))
 76    plate = ShellMesh(vertices=[[0, 0, 0], [lengthX, 0, 0], [lengthX, lengthY, 0], [0, lengthY, 0]],
 77                      numberOfElementsX=nx, numberOfElementsY=ny, youngsModulus=youngsModulus,
 78                      poissonsRatio=poissonRatio, density=density, thickness=stripThickness,
 79                      stiffnessProportionalDamping=stiffnessProportionalDamping)
 80    if thicknessField is not None:
 81        plate.thicknessAtNodes = [thicknessField(ix * lengthX / nx, iy * lengthY / ny)
 82                                  for iy in range(ny + 1) for ix in range(nx + 1)]
 83    plate.CreateANCFThinPlateElements(mbs, useReducedOrderIntegration=integrationMode)
 84    if massProportionalLoad is not None:
 85        for elementNumber in plate.elementNumbers:
 86            mbs.AddLoad(LoadMassProportional(markerNumber=mbs.AddMarker(MarkerBodyMass(bodyNumber=elementNumber)),
 87                                             loadVector=massProportionalLoad))
 88    built = {'nodes': plate.nodeNumbers, 'elements': plate.elementNumbers,
 89             'edgeNodeNumbers': plate.boundaryNodeNumbers, 'nodeId': lambda iy, ix: iy * (nx + 1) + ix}
 90    if alongY:
 91        clampNodes, tipNodes = built['edgeNodeNumbers']['bottom'], built['edgeNodeNumbers']['top']
 92    else:
 93        clampNodes, tipNodes = built['edgeNodeNumbers']['left'], built['edgeNodeNumbers']['right']
 94    if clamp:
 95        for nodeNumber in clampNodes:
 96            AddNodeConstraints(mbs, nodeNumber, dofs=list(range(numberOfPlateDOFsPerNode)))
 97    return systemContainer, mbs, built, tipNodes

100def AddTipLoad(mbs, tipNodes, force=tipForce):
101    """Distribute a force in -z equally on the tip nodes.
102    @param mbs: main system
103    @param tipNodes: list of tip node numbers
104    @param force: total force (N)"""
105    for nodeNumber in tipNodes:
106        markerNumber = mbs.AddMarker(MarkerNodePosition(nodeNumber=nodeNumber))
107        mbs.AddLoad(LoadForceVector(markerNumber=markerNumber, loadVector=[0, 0, -force / len(tipNodes)]))

110def TipDeflection(mbs, tipNodes):
111    """Return the mean downward (-z) displacement of the tip nodes.
112    @param mbs: main system (solved)
113    @param tipNodes: list of tip node numbers
114    @return: deflection (m), positive downward"""
115    return -np.mean([mbs.GetNodeOutput(nodeNumber, exu.OutputVariableType.Displacement)[2] for nodeNumber in tipNodes])

118def RigidMotionVectors(mbs, rotationMatrix, angularVelocity):
119    """Coordinates of a rigid rotation of all plate nodes and the velocities of a rigid angular velocity in
120    this rotated configuration (positions and both slope vectors rotate with the body).
121    @param mbs: main system (assembled, NodePointSlope12 nodes only)
122    @param rotationMatrix: 3x3 rotation of the reference configuration
123    @param angularVelocity: angular velocity vector (rad/s)
124    @return: (ODE2 coordinates, ODE2 velocities) as numpy arrays"""
125    coordinates = np.zeros(mbs.systemData.ODE2Size())
126    velocities = np.zeros(mbs.systemData.ODE2Size())
127    for nodeNumber in range(mbs.systemData.NumberOfNodes()):
128        referenceCoordinates = np.array(mbs.GetNodeParameter(nodeNumber, 'referenceCoordinates'))
129        firstIndex = mbs.GetNodeODE2Index(nodeNumber)
130        for vectorIndex in range(3):
131            referenceVector = referenceCoordinates[3 * vectorIndex:3 * vectorIndex + 3]
132            rotatedVector = rotationMatrix @ referenceVector
133            coordinates[firstIndex + 3 * vectorIndex:firstIndex + 3 * vectorIndex + 3] = rotatedVector - referenceVector
134            velocities[firstIndex + 3 * vectorIndex:firstIndex + 3 * vectorIndex + 3] = np.cross(angularVelocity, rotatedVector)
135    return coordinates, velocities

138def PrintComparison(label, measured, reference):
139    """Verbose test output: measured value, reference value and relative deviation.

141    @param measured: measured value
142    @param reference: reference value (nonzero)"""

144          f'relative deviation {measured / reference - 1:+.3e}', end='')

150    def testCantileverTipDeflectionAllIntegrationModes(self):
151        """Linear cantilever with tip load: cubic Hermite elements are exact, F L^3/(3 EI), for modes 0, 1, 2;
152        mode 2 is an alias of mode 1 (bitwise identical)."""
153        deflections = {}
154        for integrationMode in [0, 1, 2]:
155            systemContainer, mbs, built, tipNodes = CreateStrip(numberOfElements=2, integrationMode=integrationMode)
156            AddTipLoad(mbs, tipNodes)
157            mbs.Assemble()
158            mbs.SolveStatic(StaticSettings())
159            deflections[integrationMode] = TipDeflection(mbs, tipNodes)
160            reference = tipForce * stripLength**3 / (3 * bendingStiffness)
161            PrintComparison(f'tip deflection mode {integrationMode}', deflections[integrationMode], reference)
162            self.assertAlmostEqual(deflections[integrationMode] / reference, 1., delta=1e-6)
163        self.assertEqual(deflections[1], deflections[2])

179    def testMaterialCurvatureOfStretchedArc(self):
180        """Material curvature measure: a strip placed exactly on a circular arc of radius rho and stretched by
181        lambda has |kappa_xx| = lambda/rho (geometric measure: 1/rho) and Green-Lagrange strain (lambda^2-1)/2.
182        Pure kinematics, no solver."""
183        stretch, radius = 1.2, 1.0
184        systemContainer, mbs, built, tipNodes = CreateStrip(numberOfElements=4, clamp=False)
185        mbs.Assemble()
186        coordinates = np.zeros(mbs.systemData.ODE2Size())
187        for nodeNumber in range(mbs.systemData.NumberOfNodes()):
188            referenceCoordinates = np.array(mbs.GetNodeParameter(nodeNumber, 'referenceCoordinates'))
189            arcAngle = stretch * referenceCoordinates[0] / radius
190            position = np.array([radius * np.sin(arcAngle), referenceCoordinates[1], radius * (1 - np.cos(arcAngle))])
191            slopeX = stretch * np.array([np.cos(arcAngle), 0, np.sin(arcAngle)])
192            firstIndex = mbs.GetNodeODE2Index(nodeNumber)
193            coordinates[firstIndex:firstIndex + 3] = position - referenceCoordinates[0:3]
194            coordinates[firstIndex + 3:firstIndex + 6] = slopeX - referenceCoordinates[3:6]
195        mbs.systemData.SetODE2Coordinates(coordinates.tolist(), configuration=exu.ConfigurationType.Current)
196        for elementNumber in built['elements']:
197            curvature = mbs.GetObjectOutputBody(elementNumber, exu.OutputVariableType.CurvatureLocal,
198                                                localPosition=[0, 0, 0], configuration=exu.ConfigurationType.Current)
199            strain = mbs.GetObjectOutputBody(elementNumber, exu.OutputVariableType.StrainLocal,
200                                             localPosition=[0, 0, 0], configuration=exu.ConfigurationType.Current)
201            PrintComparison(f'element {elementNumber} |kappa_xx|', abs(curvature[0]), stretch / radius)
202            self.assertAlmostEqual(abs(curvature[0]) / (stretch / radius), 1., delta=1e-2)   #interpolation error 0.4 %
203            self.assertAlmostEqual(strain[0], 0.5 * (stretch**2 - 1), delta=1e-4)
204            self.assertAlmostEqual(curvature[1], 0., delta=1e-10)

206    def testPureBendingLargeRotation(self):
207        """Tip slope rotated by 1 rad: the strip forms a circular arc of radius L/theta; curvature theta/L,
208        membrane strain small compared with the surface bending strain."""
209        rotationAngle = 1.0
210        systemContainer, mbs, built, tipNodes = CreateStrip(numberOfElements=4)
211        groundNode = mbs.AddNode(NodePointGround())
212        targetSlope = [np.cos(rotationAngle), 0., np.sin(rotationAngle)]
213        for nodeNumber in tipNodes:
214            referenceCoordinates = mbs.GetNodeParameter(nodeNumber, 'referenceCoordinates')
215            for component in range(3):
216                groundMarker = mbs.AddMarker(MarkerNodeCoordinate(nodeNumber=groundNode, coordinate=0))
217                slopeMarker = mbs.AddMarker(MarkerNodeCoordinate(nodeNumber=nodeNumber, coordinate=3 + component))
218                mbs.AddObject(CoordinateConstraint(markerNumbers=[groundMarker, slopeMarker],
219                                                   offset=targetSlope[component] - referenceCoordinates[3 + component]))
220        mbs.Assemble()
221        mbs.SolveStatic(StaticSettings(numberOfBendingLoadSteps))
222        tipPosition = np.mean([mbs.GetNodeOutput(nodeNumber, exu.OutputVariableType.Position) for nodeNumber in tipNodes], axis=0)
223        exactTipPosition = stripLength / rotationAngle * np.array([np.sin(rotationAngle), 1 - np.cos(rotationAngle)])
224        arcError = np.linalg.norm(tipPosition[[0, 2]] - exactTipPosition)
225        print(f'\n    tip position error {arcError:.2e} m', end='')
226        self.assertLess(arcError, 1e-5)
227        for elementNumber in built['elements']:
228            curvature = mbs.GetObjectOutputBody(elementNumber, exu.OutputVariableType.CurvatureLocal, localPosition=[0, 0, 0])
229            membraneStrain = mbs.GetObjectOutputBody(elementNumber, exu.OutputVariableType.StrainLocal, localPosition=[0, 0, 0])
230            surfaceStrain = mbs.GetObjectOutputBody(elementNumber, exu.OutputVariableType.StrainLocal, localPosition=[0, 0, 1])
231            PrintComparison(f'element {elementNumber} |kappa_xx|', abs(curvature[0]), rotationAngle / stripLength)
232            self.assertAlmostEqual(abs(curvature[0]) / (rotationAngle / stripLength), 1., delta=5e-3)
233            self.assertLess(abs(membraneStrain[0] / surfaceStrain[0]), 2e-3)

235    def testFirstEigenfrequencyCantilever(self):

285    def testLinearTaperTipDeflection(self):
286        """Thickness h(x) = h0 (1 - x/(2L)), bending stiffness point-wise with h^3: tip deflection
287        integral of F (L-x)^2 / (E b h(x)^3/12) dx."""
288        thicknessFunction = lambda x, y: stripThickness * (1 - 0.5 * x / stripLength)
289        systemContainer, mbs, built, tipNodes = CreateStrip(numberOfElements=4, thicknessField=thicknessFunction)
290        AddTipLoad(mbs, tipNodes)
291        mbs.Assemble()
292        mbs.SolveStatic(StaticSettings())
293        positions = np.linspace(0, stripLength, 20001)
294        integrand = tipForce * (stripLength - positions)**2 / (youngsModulus * stripWidth * thicknessFunction(positions, 0)**3 / 12)
295        reference = np.sum(0.5 * (integrand[1:] + integrand[:-1]) * np.diff(positions))   #trapezoidal rule
296        PrintComparison('tip deflection, linear taper', TipDeflection(mbs, tipNodes), reference)

314class TestANCFThinPlateDamping(unittest.TestCase):
315    """Kelvin-Voigt damping, checked with the damping matrix of the linearized system."""

317    def DampingMatrix(self, membraneDamping, bendingDamping, coordinates=None):
318        """Damping matrix of a free 2-element strip.
319        @param membraneDamping: stiffnessProportionalDamping (s)
320        @param bendingDamping: bendingStiffnessProportionalDamping (s), negative: use membraneDamping
321        @param coordinates: optional ODE2 coordinates (initial and current configuration)
322        @return: (mbs, stiffness matrix, damping matrix)"""
323        systemContainer, mbs, built, tipNodes = CreateStrip(numberOfElements=2, clamp=False,
324                                                            stiffnessProportionalDamping=membraneDamping)
325        for elementNumber in built['elements']:
326            mbs.SetObjectParameter(elementNumber, 'bendingStiffnessProportionalDamping', bendingDamping)
327        mbs.Assemble()
328        if coordinates is not None:
329            for configuration in [exu.ConfigurationType.Initial, exu.ConfigurationType.Current]:
330                mbs.systemData.SetODE2Coordinates(coordinates.tolist(), configuration=configuration)
331        massMatrix, stiffnessMatrix, dampingMatrix = mbs.ComputeLinearizedSystem(StaticSettings())
332        return mbs, stiffnessMatrix, dampingMatrix

334    def testNoDampingOfRigidRotation(self):
335        """In a finitely rotated configuration, a rigid angular velocity produces no damping force (strain
336        rates vanish), while a bending velocity does; also no elastic force change for rigid rotation."""
337        rotationAngle = 1.0
338        rotationMatrix = np.array([[np.cos(rotationAngle), 0, np.sin(rotationAngle)], [0, 1, 0],
339                                   [-np.sin(rotationAngle), 0, np.cos(rotationAngle)]])
340        mbs, stiffnessMatrix, dampingMatrix = self.DampingMatrix(1e-3, -1.)
341        coordinates, rigidVelocities = RigidMotionVectors(mbs, rotationMatrix, np.array([0.3, -0.5, 0.7]))
342        mbs, stiffnessMatrix, dampingMatrix = self.DampingMatrix(1e-3, -1., coordinates)
343        bendingVelocities = np.zeros(mbs.systemData.ODE2Size())
344        for nodeNumber in range(mbs.systemData.NumberOfNodes()):
345            referenceCoordinates = np.array(mbs.GetNodeParameter(nodeNumber, 'referenceCoordinates'))
346            firstIndex = mbs.GetNodeODE2Index(nodeNumber)
347            bendingVelocities[firstIndex:firstIndex + 3] = rotationMatrix @ np.array([0, 0, referenceCoordinates[0]**2])
348        rigidForce = np.linalg.norm(dampingMatrix @ rigidVelocities)
349        bendingForce = np.linalg.norm(dampingMatrix @ bendingVelocities)
350        print(f'\n    |D v_rigid| = {rigidForce:.3e}, |D v_bending| = {bendingForce:.3e}, '
351              f'|K v_rigid|/|K v_bending| = {np.linalg.norm(stiffnessMatrix @ rigidVelocities) / np.linalg.norm(stiffnessMatrix @ bendingVelocities):.3e}', end='')
352        self.assertGreater(bendingForce, 1.)
353        self.assertLess(rigidForce / bendingForce, 1e-9)
354        self.assertLess(np.linalg.norm(stiffnessMatrix @ rigidVelocities) / np.linalg.norm(stiffnessMatrix @ bendingVelocities), 1e-9)

356    def testBendingDampingParameterRule(self):
357        """Flat strip: out-of-plane velocity w = x^2 only strains in bending, in-plane velocity u = x only in the
358        membrane. beta_kappa >= 0 is used as is (0 switches bending damping off), negative uses beta_epsilon."""
359        cases = [(1e-3, 0., False, True), (1e-3, -1., True, True), (0., 1e-3, True, False), (0., -1., False, False)]
360        for membraneDamping, bendingDamping, bendingDamped, membraneDamped in cases:
361            mbs, stiffnessMatrix, dampingMatrix = self.DampingMatrix(membraneDamping, bendingDamping)
362            bendingVelocities = np.zeros(mbs.systemData.ODE2Size())
363            stretchVelocities = np.zeros(mbs.systemData.ODE2Size())
364            for nodeNumber in range(mbs.systemData.NumberOfNodes()):
365                referenceCoordinates = np.array(mbs.GetNodeParameter(nodeNumber, 'referenceCoordinates'))
366                firstIndex = mbs.GetNodeODE2Index(nodeNumber)
367                bendingVelocities[firstIndex + 2] = referenceCoordinates[0]**2          #w = x^2
368                bendingVelocities[firstIndex + 5] = 2 * referenceCoordinates[0]         #w_x = 2x
369                stretchVelocities[firstIndex] = referenceCoordinates[0]                 #u = x
370                stretchVelocities[firstIndex + 3] = 1.                                  #u_x = 1
371            bendingForce = np.linalg.norm(dampingMatrix @ bendingVelocities)
372            stretchForce = np.linalg.norm(dampingMatrix @ stretchVelocities)
373            print(f'\n    beta_epsilon = {membraneDamping}, beta_kappa = {bendingDamping}: '
374                  f'|D v_bending| = {bendingForce:.3e}, |D v_stretch| = {stretchForce:.3e}', end='')
375            self.assertEqual(bendingForce > 1e-6, bendingDamped)
376            self.assertEqual(stretchForce > 1e-6, membraneDamped)

490    def testThicknessFunctionOfARotatedStrip(self):
491        """A strip turned by 30 degrees in the x-y plane with the thickness h0 (1 - s/(2L)) along its axis s, given as a
492        function of the global x and y: the gradients along the node slopes give 12 thickness values per element, and
493        the tip deflection is the one of the tapered cantilever (#2864)."""
494        angle = np.pi / 6
495        axis, across = np.array([np.cos(angle), np.sin(angle), 0.]), np.array([-np.sin(angle), np.cos(angle), 0.])
496        def ThicknessFunction(x, y):
497            return stripThickness * (1 - 0.5 * (x * axis[0] + y * axis[1]) / stripLength)
498        systemContainer = exu.SystemContainer()
499        mbs = systemContainer.AddSystem()
500        plate = ShellMesh(vertices=[[0, 0, 0], stripLength * axis, stripLength * axis + stripWidth * across, stripWidth * across],
501                          numberOfElementsX=8, numberOfElementsY=1, youngsModulus=youngsModulus, poissonsRatio=poissonRatio,
502                          density=density, thickness=stripThickness, thicknessFunction=ThicknessFunction)
503        plate.CreateANCFThinPlateElements(mbs)
504        self.assertEqual(len(mbs.GetObjectParameter(plate.elementNumbers[0], 'thickness')), 12)
505        for nodeNumber in plate.boundaryNodeNumbers['left']:
506            AddNodeConstraints(mbs, nodeNumber, dofs=list(range(numberOfPlateDOFsPerNode)))
507        tipNodes = plate.boundaryNodeNumbers['right']
508        for nodeNumber in tipNodes:
509            mbs.AddLoad(LoadForceVector(markerNumber=mbs.AddMarker(MarkerNodePosition(nodeNumber=nodeNumber)),
510                                        loadVector=[0, 0, -tipForce / len(tipNodes)]))
511        mbs.Assemble()
512        mbs.SolveStatic(StaticSettings())
513        positions = np.linspace(0, stripLength, 20001)
514        thickness = stripThickness * (1 - 0.5 * positions / stripLength)
515        integrand = tipForce * (stripLength - positions)**2 / (youngsModulus * stripWidth * thickness**3 / 12)
516        reference = np.sum(0.5 * (integrand[1:] + integrand[:-1]) * np.diff(positions))   #trapezoidal rule
517        PrintComparison('tip deflection, thickness function of a rotated strip', TipDeflection(mbs, tipNodes), reference)
