 13import exudyn as exu
 14#a local run must not open a plot window when windows are suppressed (#2477)
 15if exu.special.userInterface.suppressPlots: import matplotlib; matplotlib.use('Agg')
 16from exudyn.utilities import * #includes itemInterface and rigidBodyUtilities
 17import exudyn.graphics as graphics
 18from exudyn.FEM import *

 20import numpy as np

 22testIsActive = exu.sys.get('testIsActive', False)

 24SC = exu.SystemContainer()
 25mbs = SC.AddSystem()
 26testIsActive = True #without test

 28
 29#%%+++++++++++++++++++++++++++++++++++++++++++++++++++++
 30#Use FEMinterface to import FEM model and create FFRFreducedOrder object
 31fem = FEMinterface()
 32#inputFileName = 'C:/DATA/cpp/EXUDYN_git/main/pythonDev/TestModels/testData/rotorDiscTest' #runTestSuite.py is at another directory
 33inputFileName = 'testData/rotorDiscTest' #runTestSuite.py is at another directory
 34#if not testIsActive:
 35#    inputFileName = 'testData/rotorDiscTest'        #if executed in current directory
 36
 37nodes=fem.ImportFromAbaqusInputFile(inputFileName+'.inp', typeName='Instance', name='rotor-1')
 38
 39fem.ReadMassMatrixFromAbaqus(inputFileName+'MASS1.mtx')
 40fem.ReadStiffnessMatrixFromAbaqus(inputFileName+'STIF1.mtx')
 41fem.ScaleStiffnessMatrix(6e-4) #for larger deformations, stiffness is reduced to 1%
 42
 43#nodeNumberUnbalance = 9  #on disc, max y-value
 44nodeNumberUnbalance = fem.GetNodeAtPoint(point=[0. , 0.19598444, 0.15])
 45#exu.Print("nodeNumberUnbalance =",nodeNumberUnbalance)
 46unbalance = 0.1
 47fem.AddNodeMass(nodeNumberUnbalance, unbalance)
 48
 49nModes = 8
 50fem.ComputeEigenmodes(nModes, excludeRigidBodyModes = 6, useSparseSolver = False)
 51#print("eigen freq.=", fem.GetEigenFrequenciesHz())
 52
 53cms = ObjectFFRFreducedOrderInterface(fem)
 54
 55#user functions should be defined outside of class:
 56def UFmassFFRFreducedOrder(mbs, t, itemIndex, qReduced, qReduced_t):
 57    return cms.UFmassFFRFreducedOrder(exu, mbs, t, qReduced, qReduced_t)
 58
 59def UFforceFFRFreducedOrder(mbs, t, itemIndex, qReduced, qReduced_t):
 60    return cms.UFforceFFRFreducedOrder(exu, mbs, t, qReduced, qReduced_t)
 61
 62objFFRF = cms.AddObjectFFRFreducedOrderWithUserFunctions(exu, mbs, positionRef=[0,0,0], rotationMatrixRef=np.eye(3), 
 63                                              initialVelocity=[0,0,0], initialAngularVelocity=[0,0,50*2*pi],
 64                                              gravity = [0,-0*9.81,0],
 65                                              UFforce=UFforceFFRFreducedOrder, UFmassMatrix=UFmassFFRFreducedOrder,
 66                                              color=[0.1,0.9,0.1,1.])
 67
 68#%%+++++++++++++++++++++++++++++++++++++++++++++++++++++
 69#add markers and joints
 70nodeDrawSize = 0.0025 #for joint drawing
 71
 72pLeft = [0,0,0]
 73pRight = [0,0,0.5]
 74nMid = fem.GetNodeAtPoint([0,0.05,0.25])
 75#exu.Print("nMid=",nMid)
 76
 77mRB = mbs.AddMarker(MarkerNodeRigid(nodeNumber=objFFRF['nRigidBody']))
 78oGround = mbs.AddObject(ObjectGround(referencePosition= [0,0,0]))
 79
 80mGroundPosLeft = mbs.AddMarker(MarkerBodyPosition(bodyNumber=oGround, localPosition=pLeft))
 81mGroundPosRight = mbs.AddMarker(MarkerBodyPosition(bodyNumber=oGround, localPosition=pRight))
 82
 83#torque on reference frame:
 84#mbs.AddLoad(Torque(markerNumber=mRB, loadVector=[0,0,100*2*pi])) 
 85
 86if False: #OPTIONAL: lock rigid body motion of reference frame (for tests):
 87    mbs.AddObject(GenericJoint(markerNumbers=[mGround, mRB], constrainedAxes=[1,1,1, 1,1,0]))
 88
 89#++++++++++++++++++++++++++++++++++++++++++
 90#find nodes at left and right surface:
 91nodeListLeft = fem.GetNodesInPlane(pLeft, [0,0,1])
 92nodeListRight = fem.GetNodesInPlane(pRight, [0,0,1])
 93#nLeft = fem.GetNodeAtPoint(pLeft)
 94#nRight = fem.GetNodeAtPoint(pRight)
 95
 96
 97lenLeft = len(nodeListLeft)
 98lenRight = len(nodeListRight)
 99weightsLeft = np.array((1./lenLeft)*np.ones(lenLeft))
100weightsRight = np.array((1./lenRight)*np.ones(lenRight))
101
102addSupports = True
103if addSupports:
104    k = 2e8     #joint stiffness
105    d = k*0.01  #joint damping
106
107    useSpringDamper = True
108
109    mLeft = mbs.AddMarker(MarkerSuperElementPosition(bodyNumber=objFFRF['oFFRFreducedOrder'], 
110                                                    meshNodeNumbers=np.array(nodeListLeft), #these are the meshNodeNumbers
111                                                    weightingFactors=weightsLeft))
112    mRight = mbs.AddMarker(MarkerSuperElementPosition(bodyNumber=objFFRF['oFFRFreducedOrder'], 
113                                                    meshNodeNumbers=np.array(nodeListRight), #these are the meshNodeNumbers 
114                                                    weightingFactors=weightsRight))
115    if useSpringDamper:
116        oSJleft = mbs.AddObject(CartesianSpringDamper(markerNumbers=[mLeft, mGroundPosLeft],
117                                            stiffness=[k,k,k], damping=[d,d,d]))
118        oSJright = mbs.AddObject(CartesianSpringDamper(markerNumbers=[mRight,mGroundPosRight],
119                                            stiffness=[k,k,0], damping=[d,d,d]))
120    else:
121        oSJleft = mbs.AddObject(SphericalJoint(markerNumbers=[mGroundPosLeft,mLeft], visualization=VObjectJointSpherical(jointRadius=nodeDrawSize)))
122        oSJright= mbs.AddObject(SphericalJoint(markerNumbers=[mGroundPosRight,mRight], visualization=VObjectJointSpherical(jointRadius=nodeDrawSize)))
123

126fileDir = 'solution/'

128mbs.AddSensor(SensorNode(nodeNumber=objFFRF['nRigidBody'], storeInternal=True,#fileName=fileDir+'rbDisplacement.txt', 
129                         outputVariableType = exu.OutputVariableType.Displacement))

131mbs.AddSensor(SensorNode(nodeNumber=objFFRF['nRigidBody'], storeInternal=True,#fileName=fileDir+'rbVelocity.txt', 
132                         outputVariableType = exu.OutputVariableType.Velocity))

137mbs.AddSensor(SensorNode(nodeNumber=objFFRF['nRigidBody'], storeInternal=True,#fileName=fileDir+'nRigidBodyAngVelCMS.txt', 
138                         outputVariableType = exu.OutputVariableType.AngularVelocity))

141#FFRF object, selected node:
142sCMSdisp=mbs.AddSensor(SensorSuperElement(bodyNumber=objFFRF['oFFRFreducedOrder'], meshNodeNumber=nMid, #meshnode number!
143                         storeInternal=True,#fileName=fileDir+'nMidDisplacementCMS.txt', 
144                         outputVariableType = exu.OutputVariableType.Displacement))

146sCMSvel=mbs.AddSensor(SensorSuperElement(bodyNumber=objFFRF['oFFRFreducedOrder'], meshNodeNumber=nMid, #meshnode number!
147                         storeInternal=True,#fileName=fileDir+'nMidVelocityCMS.txt', 
148                         outputVariableType = exu.OutputVariableType.Velocity))
149
150sCMSacc=mbs.AddSensor(SensorSuperElement(bodyNumber=objFFRF['oFFRFreducedOrder'], meshNodeNumber=nMid, #meshnode number!
151                         storeInternal=True,#fileName=fileDir+'nMidAccelerationCMS.txt', 
152                         outputVariableType = exu.OutputVariableType.Acceleration))

154mbs.Assemble()
