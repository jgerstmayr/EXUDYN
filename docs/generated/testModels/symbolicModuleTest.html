430def ExtendedSymbolicChecks():
431    wrong = 0
432
433    def Check(condition, info):
434        if not condition:
435            exu.Print('*** wrong: ' + info)
436        return 0 if condition else 1
437
438    #--- Diff: compared against a central difference of the same expression
439    esym.SetRecording(True)
440    for value in [0.3, 1.7, -0.8]:
441        x = SymReal('x', value)
442        cases = [(x*x*x, 3*value**2),
443                 (esym.sin(x)*esym.cos(x), math.cos(2*value)),
444                 (esym.exp(x)/x, math.exp(value)/value - math.exp(value)/value**2),
445                 (esym.sqrt(x*x+1), value/math.sqrt(value**2+1)),
446                 ]
447        for (f, analytic) in cases:
448            wrong += Check(abs(f.Diff(x) - analytic) < 1e-12, 'Diff value ' + str(value))
